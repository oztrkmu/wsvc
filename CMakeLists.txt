cmake_minimum_required(VERSION 3.20)
project(wsvc VERSION 0.1.0 LANGUAGES C)
if(NOT WIN32)
    message(FATAL_ERROR "wsvc targets Windows. Use Windows or a MinGW toolchain.")
endif()
add_library(wsvc STATIC src/wsvc.c src/wautorun.c)
target_include_directories(wsvc PUBLIC include)
target_compile_features(wsvc PUBLIC c_std_11)
target_compile_definitions(wsvc PUBLIC UNICODE _UNICODE _WIN32_WINNT=0x0600)
target_link_libraries(wsvc PUBLIC advapi32)
add_executable(wsvc-example example/main.c)
target_link_libraries(wsvc-example PRIVATE wsvc)
if(MINGW)
    target_link_options(wsvc-example PRIVATE -municode)
endif()
foreach(target wsvc wsvc-example)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
    add_executable(wsvc-api-test tests/api.c)
    target_link_libraries(wsvc-api-test PRIVATE wsvc)
    add_test(NAME api-validation COMMAND wsvc-api-test)
    add_executable(wautorun-test tests/autorun.c)
    target_link_libraries(wautorun-test PRIVATE wsvc)
    add_test(NAME autorun-registry COMMAND wautorun-test)
    add_test(NAME cli-help COMMAND wsvc-example --help)
    add_test(NAME cli-invalid COMMAND wsvc-example --unknown)
    set_tests_properties(cli-invalid PROPERTIES WILL_FAIL TRUE)
    add_test(NAME cli-invalid-startup COMMAND wsvc-example --install invalid)
    set_tests_properties(cli-invalid-startup PROPERTIES WILL_FAIL TRUE)
endif()
